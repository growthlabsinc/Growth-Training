# Story 11.6 Implementation Plan: Automated Weekly GrowthTrack Export

## Overview

This document outlines the implementation plan for **Story 11.6: Automated Weekly GrowthTrack Export**, which enables a **hybrid approach** for CSV data export to Karl's GrowthTrack research platform.

## Hybrid Export Strategy

### Two Export Options for Users

**1. Manual Export (Story 11.2 - Already Complete)**
- User taps "Export My Data" button in Settings
- CSVExportService generates CSV on-demand
- iOS share sheet presents sharing options
- User decides how/when to share with Karl
- **Best for:** Users who want full control

**2. Automated Weekly Export (Story 11.6 - To Be Implemented)**
- User opts-in via toggle: "Automatically share data with GrowthTrack (weekly)"
- Firebase Cloud Function runs every Sunday at midnight UTC
- Aggregates all opted-in users' data into single CSV
- Uploads to Cloud Storage and emails download link to Karl
- **Best for:** Users who want to contribute passively

### Users Can Choose Both

The toggle is independent - users can:
- Use manual export only (default)
- Use automated export only
- Use both (export manually AND contribute to weekly batches)

```javascript
const functions = require('firebase-functions');
const admin = require('firebase-admin');
const sgMail = require('@sendgrid/mail');

exports.weeklyGrowthTrackExport = functions.pubsub
  .schedule('0 0 * * 0') // Every Sunday at midnight UTC
  .timeZone('America/Los_Angeles')
  .onRun(async (context) => {
    console.log('🚀 Starting weekly GrowthTrack export...');

    try {
      // 1. Query opted-in users
      const usersSnapshot = await admin.firestore()
        .collection('users')
        .where('growthTrackOptIn', '==', true)
        .get();

      if (usersSnapshot.empty) {
        console.log('⚠️  No opted-in users found');
        return null;
      }

      console.log(`✅ Found ${usersSnapshot.size} opted-in users`);

      // 2. Fetch sessions for each user (only new since last export)
      const allSessions = [];
      for (const userDoc of usersSnapshot.docs) {
        const userId = userDoc.id;
        const userData = userDoc.data();
        const lastExported = userData.lastExportedAt?.toDate() || new Date(0);

        const sessionsSnapshot = await admin.firestore()
          .collection('sessionLogs')
          .where('userId', '==', userId)
          .where('startTime', '>', lastExported)
          .get();

        console.log(`  User ${userId}: ${sessionsSnapshot.size} new sessions`);

        sessionsSnapshot.docs.forEach(doc => {
          const session = doc.data();
          // Get anonymous ID for this user
          const anonymousId = userData.anonymousStatisticalId || generateAnonymousId();

          allSessions.push({
            anonymousId,
            ...session
          });
        });
      }

      if (allSessions.length === 0) {
        console.log('⚠️  No new sessions to export');
        return null;
      }

      console.log(`✅ Total sessions to export: ${allSessions.length}`);

      // 3. Generate CSV (reuse Story 11.2 logic)
      const csvContent = generateCSVContent(allSessions);

      // 4. Upload to Cloud Storage
      const dateStr = new Date().toISOString().split('T')[0];
      const fileName = `growthtrack-export-${dateStr}.csv`;
      const bucket = admin.storage().bucket();
      const file = bucket.file(`growthtrack-exports/${fileName}`);

      await file.save(csvContent, {
        contentType: 'text/csv',
        metadata: {
          userCount: usersSnapshot.size,
          sessionCount: allSessions.length,
          exportDate: dateStr
        }
      });

      console.log(`✅ CSV uploaded: ${fileName}`);

      // 5. Generate signed URL (expires in 7 days)
      const [url] = await file.getSignedUrl({
        action: 'read',
        expires: Date.now() + 7 * 24 * 60 * 60 * 1000
      });

      // 6. Send email via SendGrid
      await sendExportEmail({
        url,
        userCount: usersSnapshot.size,
        sessionCount: allSessions.length,
        dateRange: getDateRange(allSessions),
        fileName
      });

      console.log('✅ Email sent to Karl');

      // 7. Update user records with lastExportedAt
      const batch = admin.firestore().batch();
      const exportTimestamp = admin.firestore.FieldValue.serverTimestamp();

      usersSnapshot.docs.forEach(doc => {
        batch.update(doc.ref, { lastExportedAt: exportTimestamp });
      });

      await batch.commit();
      console.log('✅ User records updated');

      // 8. Log audit trail
      await admin.firestore().collection('growthtrack_exports').add({
        timestamp: exportTimestamp,
        exportType: 'scheduled',
        userCount: usersSnapshot.size,
        sessionCount: allSessions.length,
        csvUrl: url,
        fileName,
        success: true
      });

      console.log('✅ Export complete!');
      return null;

    } catch (error) {
      console.error('❌ Export failed:', error);

      // Send error notification to admin
      await sendErrorEmail(error);

      // Log failure
      await admin.firestore().collection('growthtrack_exports').add({
        timestamp: admin.firestore.FieldValue.serverTimestamp(),
        exportType: 'scheduled',
        success: false,
        errorMessage: error.message,
        errorStack: error.stack
      });

      throw error;
    }
  });

// Helper: Generate CSV content
function generateCSVContent(sessions) {
  const header = 'anonymous_id,date,category,duration_minutes,pre_bpel_mm,pre_bpfsl_mm,pre_mseg_mm,post_bpel_mm,post_bpfsl_mm,post_mseg_mm';

  const rows = sessions
    .filter(session => {
      // Only include sessions with complete measurements
      return session.preMeasurements?.bpel &&
             session.preMeasurements?.bpfsl &&
             session.preMeasurements?.mseg &&
             session.postMeasurements?.bpel &&
             session.postMeasurements?.bpfsl &&
             session.postMeasurements?.mseg;
    })
    .map(session => {
      const date = formatDate(session.startTime);
      const category = 'hybrid'; // TODO: Implement category mapping
      const duration = session.duration || 0;

      // Convert inches to mm (× 25.4, rounded)
      const toMm = (inches) => Math.round(inches * 25.4);

      return [
        session.anonymousId,
        date,
        category,
        duration,
        toMm(session.preMeasurements.bpel),
        toMm(session.preMeasurements.bpfsl),
        toMm(session.preMeasurements.mseg),
        toMm(session.postMeasurements.bpel),
        toMm(session.postMeasurements.bpfsl),
        toMm(session.postMeasurements.mseg)
      ].join(',');
    });

  return [header, ...rows].join('\n');
}

// Helper: Format date as YYYY-MM-DD
function formatDate(timestamp) {
  const date = timestamp.toDate ? timestamp.toDate() : new Date(timestamp);
  return date.toISOString().split('T')[0];
}

// Helper: Get date range from sessions
function getDateRange(sessions) {
  if (sessions.length === 0) return 'N/A';

  const dates = sessions.map(s => {
    const date = s.startTime.toDate ? s.startTime.toDate() : new Date(s.startTime);
    return date.getTime();
  });

  const minDate = new Date(Math.min(...dates));
  const maxDate = new Date(Math.max(...dates));

  return `${formatDate(minDate)} to ${formatDate(maxDate)}`;
}
```

**Add State & Methods:**

```swift
@State private var automatedExportEnabled: Bool = false
@State private var lastExportDate: Date? = nil

private func updateAutomatedExportSetting(_ enabled: Bool) {
    guard let userId = Auth.auth().currentUser?.uid else { return }

    let db = Firestore.firestore()
    db.collection("users").document(userId).updateData([
        "growthTrackOptIn": enabled,
        "growthTrackOptInDate": enabled ? FieldValue.serverTimestamp() : FieldValue.delete()
    ]) { error in
        if let error = error {
            print("Error updating opt-in: \(error)")
        } else {
            print("Automated export \(enabled ? "enabled" : "disabled")")
        }
    }
}

private func fetchAutomatedExportSettings() {
    guard let userId = Auth.auth().currentUser?.uid else { return }

    let db = Firestore.firestore()
    db.collection("users").document(userId).getDocument { snapshot, error in
        guard let data = snapshot?.data() else { return }

        automatedExportEnabled = data["growthTrackOptIn"] as? Bool ?? false

        if let timestamp = data["lastExportedAt"] as? Timestamp {
            lastExportDate = timestamp.dateValue()
        }
    }
}
```

## Security & Compliance

### GDPR Compliance Checklist

- ✅ **Explicit opt-in:** Default is OFF, user must actively enable
- ✅ **Clear disclosure:** Settings explains "weekly automated export" clearly
- ✅ **Right to withdraw:** User can disable anytime
- ✅ **Audit trail:** Every export logged in Firestore `growthtrack_exports` collection
- ✅ **Data minimization:** Only export necessary fields (no PII)
- ✅ **Purpose limitation:** Data used only for PE research
- ✅ **Data processor agreement:** Need written agreement with Karl/GrowthTrack
- ✅ **User access:** User can see `lastExportedAt` timestamp in Settings

### Security Measures

1. **API Key Protection:**
   - SendGrid key stored in Firebase Functions config (encrypted)
   - Never exposed in client-side code
   - Rotated regularly

2. **Data Access:**
   - Only opted-in users' data exported
   - Firestore security rules prevent unauthorized access
   - Cloud Storage bucket has restricted IAM permissions

3. **Signed URLs:**
   - 7-day expiration prevents long-term access
   - HTTPS-only
   - No public bucket access

4. **Audit Logging:**
   - Every export logged with timestamp, userCount, sessionCount
   - Failed exports logged with error details
   - Logs retained for compliance review
