### Data Models
```swift
// From existing Firebase documents
struct PEExercise {
    let id: String  // e.g., "kegel_basic", "edging_stamina"
    let title: String
    let description: String
    let category: String  // "length", "girth", "eq"
    let difficulty: String  // "beginner", "intermediate", "advanced"
    let estimatedDuration: Int  // minutes
    let equipmentNeeded: [String]
    let steps: [ExerciseStep]
}

struct ExerciseStep {
    let stepNumber: Int  // CRITICAL: Use stepNumber not order
    let title: String  // e.g., "Step 1"
    let description: String  // CRITICAL: Use description not instruction
    let tips: [String]
    let cautions: [String]
    let warnings: [String]  // Can be empty array
}
```
[Source: Firebase document structure from growth_exercises collection]
