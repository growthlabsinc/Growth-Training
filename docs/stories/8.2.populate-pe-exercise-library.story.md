### TrainingProtocol Model Structure
[Source: Growth/Core/Models/TrainingProtocol.swift]

**Required Fields for Firestore Deployment:**
```javascript
{
  // Required fields
  "id": String,              // Unique document ID (e.g., "stage1_basic_manual_stretch")
  "stage": Number,           // MUST be >= 1 for actionable protocols
  "title": String,           // Exercise name
  "description": String,     // Brief description (maps to protocolDescription in Swift)
  "instructionsText": String, // Detailed how-to instructions

  // Highly recommended fields
  "classification": String,  // "Beginner", "Intermediate", "Advanced"
  "categories": [String],    // e.g., ["length", "girth", "conditioning", "eq"]
  "equipmentNeeded": [String], // e.g., ["pump", "lube"] or [] for manual
  "estimatedDurationMinutes": Number, // e.g., 30
  "safetyNotes": String,     // CRITICAL: Must include stop signals
  "isFeatured": Boolean,     // Feature important exercises

  // Optional but useful fields
  "benefits": [String],      // Benefits of this exercise
  "relatedMethods": [String], // IDs of related exercises
  "visualPlaceholderUrl": String, // Future: image/video URL
  "progressionCriteriaText": String, // When to advance

  // Timer configuration (for timed exercises)
  "timerConfig": {
    "recommended_duration_seconds": Number,
    "is_countdown": Boolean,
    "has_intervals": Boolean,
    "intervals": [
      {
        "name": String,
        "duration_seconds": Number
      }
    ],
    "max_recommended_duration_seconds": Number
  },

  // Structured steps (optional)
  "steps": [
    {
      "step_number": Number,
      "title": String,
      "description": String,
      "duration": Number,
      "tips": [String],
      "warnings": [String],
      "intensity": String
    }
  ],

  // Timestamps
  "createdAt": Timestamp,
  "updatedAt": Timestamp
}
```

**Field Name Mapping (Firestore → Swift):**
- `description` → `protocolDescription`
- `instructionsText` → `instructionsText`
- `equipment_needed` OR `equipmentNeeded` → both accepted
- `estimated_duration_minutes` OR `estimatedDurationMinutes` → both accepted
- `safety_notes` OR `safetyNotes` → both accepted
- Timer config uses snake_case in Firestore, camelCase in Swift

**CRITICAL Firestore Field Naming:**
Use snake_case OR camelCase consistently. TrainingProtocol init handles both formats.

### Firestore Deployment Script Pattern
[Source: scripts/deploy-articles-with-disclaimers.js]

**Deployment Script Structure:**
```javascript
#!/usr/bin/env node

const admin = require('firebase-admin');

// Initialize Firebase Admin with Application Default Credentials
admin.initializeApp({
  projectId: process.env.GCLOUD_PROJECT || 'growth-training-app'
});

const db = admin.firestore();

// Exercise data array
const EXERCISES = [
  {
    id: "stage1_basic_manual_stretch",
    stage: 1,
    classification: "Beginner",
    title: "Basic Manual Stretch",
    description: "Foundational length exercise using manual stretching technique",
    instructionsText: "Detailed step-by-step instructions...",
    categories: ["length"],
    equipmentNeeded: [],
    estimatedDurationMinutes: 30,
    safetyNotes: "⚠️ MEDICAL DISCLAIMER...",
    isFeatured: true,
    benefits: ["Increases length over time", "Improves tissue flexibility"],
    createdAt: admin.firestore.FieldValue.serverTimestamp(),
    updatedAt: admin.firestore.FieldValue.serverTimestamp()
  },
  // ... more exercises
];

// Batch write to Firestore
async function deployExercises() {
  const batch = db.batch();

  EXERCISES.forEach(exercise => {
    const docRef = db.collection('growth_exercises').doc(exercise.id);
    batch.set(docRef, exercise);
  });

  await batch.commit();
  console.log(`✅ Deployed ${EXERCISES.length} exercises`);
}

deployExercises().catch(console.error);
```

**Deployment Command:**
```bash
GCLOUD_PROJECT=growth-training-app node scripts/deploy-pe-exercises.js
```
